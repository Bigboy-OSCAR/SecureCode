```python
from pathlib import Path
import re

document_paths = [
    Path("docs/ui/SCREEN_SPEC.md"),
    Path("docs/ui/USER_SCENARIOS.md"),
    Path("docs/handoffs/2.1-ui-contract-requirements.md"),
]
documents = {path: path.read_text(encoding="utf-8") for path in document_paths}
spec, scenarios, handoff = documents.values()
roadmap = Path("INTEGRATED_SYSTEM_IMPLEMENTATION_ROADMAP.md").read_text(encoding="utf-8")
screen_ids = re.findall(r"^### (SC-\d{2}) ", spec, re.M)
scenario_rows = [
    [cell.strip() for cell in line.split("|")[1:-1]]
    for line in scenarios.splitlines() if re.match(r"^\| UI-\d{2} \|", line)
]
handoff_ids = re.findall(r"^\| (H-\d{2}) \|", handoff, re.M)
scenario_ids = [row[0] for row in scenario_rows]
expected_ids = {
    "SC": {f"SC-{number:02}" for number in range(1, 10)},
    "UI": {f"UI-{number:02}" for number in range(1, 35)},
    "H": {f"H-{number:02}" for number in range(1, 11)},
}
for prefix, identifiers in zip(expected_ids, [screen_ids, scenario_ids, handoff_ids]):
    assert len(identifiers) == len(set(identifiers)), f"Duplicate {prefix} IDs"
    assert set(identifiers) == expected_ids[prefix], f"Missing {prefix} IDs"
for path, content in documents.items():
    for prefix, allowed in expected_ids.items():
        assert set(re.findall(rf"\b{prefix}-\d{{2}}\b", content)) <= allowed, path
    for target in re.findall(r"\[[^\]]+\]\(([^)]+)\)", content):
        assert (path.parent / target.split("#")[0]).is_file(), (path, target)
    in_fence = False
    table_width = None
    for line in content.splitlines():
        if line.startswith(chr(96) * 3):
            in_fence = not in_fence
        if in_fence or not line.startswith("|"):
            table_width = None
            continue
        width = len(line.split("|")[1:-1])
        assert table_width is None or width == table_width, (path, line)
        table_width = width
    assert not in_fence, f"Unclosed fence: {path}"
for screen_block in re.split(r"^### SC-\d{2} ", spec, flags=re.M)[1:]:
    assert chr(96) * 3 + "text\n" in screen_block
    assert "| 입력 |" in screen_block and "| 출력 |" in screen_block
    assert "| 인수 |" in screen_block
scenario_screens = {
    row[0]: set(re.findall(r"SC-\d{2}", row[1])) for row in scenario_rows
}
for row in scenario_rows:
    assert len(row) == 7 and all(row), row[0]
    assert re.search(r"H-\d{2}", row[5]), row[0]
    assert re.search(r"1\.\d", row[6]), row[0]
matrix_count = 0
for line in spec.splitlines():
    cells = [cell.strip() for cell in line.split("|")[1:-1]]
    if len(cells) == 10 and re.fullmatch(r"SC-\d{2}", cells[0]):
        matrix_count += 1
        for scenario_id in re.findall(r"UI-\d{2}", line):
            assert cells[0] in scenario_screens[scenario_id], (cells[0], scenario_id)
assert matrix_count == 9
contract_section = roadmap.split("### 0.6 ", 1)[1].split("### 0.7 ", 1)[0]
api_sets = []
for content in [contract_section, handoff]:
    api_sets.append({
        (method, path if path.startswith("/api/v1/") else "/api/v1" + path)
        for method, path in re.findall(r"(GET|POST|DELETE) (/[^\x60 \n]+)", content)
    })
assert api_sets[0] == api_sets[1], api_sets[0] ^ api_sets[1]
assert len(api_sets[0]) == 20
error_line = contract_section.split("필수 오류군:", 1)[1].split("\n", 1)[0]
error_codes = set(re.findall(r"\x60([A-Z_]+)\x60", error_line))
assert len(error_codes) == 16
assert all(f"\x60{code}\x60" in spec for code in error_codes)
assert all("/Users/" not in content and "/home/" not in content for content in documents.values())
print("PASS: 9 screens, 9 wireframes/input-output tables, 34 scenarios, 10 handoffs")
print("PASS: state/scenario screen mappings, ID references, local links, Markdown tables/fences")
print("PASS: 20 roadmap API routes and 16 required errors covered; no personal absolute paths")
```

## 5. limitations

- 2.1은 설계 완료 범위다. UI 렌더·build·타입·lint·행동 시험·실제 API/SDK/provider 통합 시험은 실행하지 않았다. 프로젝트 실행 코드·검사 스크립트 자체가 아직 없으며 이 단계의 필수 완료 조건도 아니다.
- 9개 와이어프레임은 텍스트 배치안이다. 실제 1280px/768px 렌더·키보드·대비 검증은 1.1 이후에 필요하다.
- JSON Schema·상태 enum·HTTP 상태·fixture·누락 조회 API는 3.2/4.2가 확정한다. H-01~H-10은 공개된 설계 요구이며 설치된 SDK 함수나 확정 실행 계약이 아니다.
- UI-01~UI-34는 이후 실행할 인수 시나리오다. 설계 검사 통과를 34개 제품 시험 통과라고 기록하지 않는다.
- 코드 예시/제품 소스 생성·Figma 파일 생성 없이 요청된 문서 범위를 수행했다. Git commit/push는 실행하지 않았다.
