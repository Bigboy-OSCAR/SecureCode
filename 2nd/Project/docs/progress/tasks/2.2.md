    return samples[name + ".valid.json"]["data"]

    return json.loads(read("contracts/" + name))

def check(ok, detail):
    if not ok:

def indexed(pairs, expected, label):
    pairs = list(pairs)
    result = dict(pairs)
    check(len(pairs) == len(result) and result.keys() == expected, f"{label}: duplicate, missing or extra ID")
    return result

def ids(prefix, count):
    return {f"{prefix}-{i:02d}" for i in range(1, count + 1)}

for name, body in [("wire", wire), ("behavior", ui)]:
    indexed(((s, True) for s in re.findall(r"^### (SC-\d+)\b", body, re.M)), screens, name)

    part = re.split(r"^##+ ", ui.split("### " + s, 1)[1], maxsplit=1, flags=re.M)[0]
    check(f"UI_BEHAVIOR.md#{s.lower()}" in wire and f'<a id="{s.lower()}"></a>' in ui, s + " anchor")
    check(all(mark in part for mark in ("목적:", "표시 필드·API 출처", "입력 규칙·버튼 활성·조작", "접근성:", "| " + s + "-A")), s + " detail")
    check(any(word in part for word in ("실패", "오류", "만료")) and "복구" in part, s + " failure/recovery")

base_rows = indexed(re.findall(r"^\| (UI-\d+) \| ([^|]+)\|", original, re.M), case_ids, "2.1 cases")

links_by_case = {}
