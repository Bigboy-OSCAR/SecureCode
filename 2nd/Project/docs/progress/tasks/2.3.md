"""Check review traceability and local evidence; does not test product behavior."""
import json
import re
from pathlib import Path
from urllib.parse import unquote, urlsplit

root = Path.cwd()
acceptance = root / "docs/ui/ACCEPTANCE.md"
handoff = root / "docs/handoffs/2.3-ui-review-fixes.md"
body = acceptance.read_text(encoding="utf-8")
requests = handoff.read_text(encoding="utf-8")
rows = re.findall(r"^\| AC-(\d{2}) / UI-(\d{2}) / (.+)$", body, re.M)
assert len(rows) == 34 and {a for a, _, _ in rows} == {f"{i:02d}" for i in range(1, 35)}
assert all(a == u and len(rest.split(" | ")) == 7 for a, u, rest in rows)
assert len(re.findall(r"^\| SC-\d{2} \|", body, re.M)) == 9

def case_ids(value):
    pairs = re.findall(r"(?:AC-|UI-)?(\d{2})(?:~(?:AC-|UI-)?(\d{2}))?", value)
    return {i for start, end in pairs for i in range(int(start), int(end or start) + 1)}

review_map = {}
for line in body.splitlines():
    if re.match(r"\| UI23-\d{2} \|", line):
        cells = [cell.strip() for cell in line.strip("|").split("|")]
        review_map[cells[0]] = case_ids(cells[1])
handoff_map = {}
for line in requests.splitlines():
    if re.match(r"\| UI23-\d{2} \|", line):
        cells = [cell.strip() for cell in line.strip("|").split("|")]
        assert cells[2] in {"P1", "P2"} and "김민지" in cells[3] and cells[4] == "OPEN"
        handoff_map[cells[0]] = case_ids(cells[1].split("; ")[1])
assert review_map == handoff_map and len(review_map) == 9
assert all(f'<a id="ui23-{i:02d}"></a>' in requests for i in range(1, 10))

paths = [acceptance, handoff, root / "docs/progress/tasks/2.3.md",
         root / "docs/progress/evidence/2.3/README.md",
         root / "docs/ui/SCREEN_SPEC.md", root / "docs/ui/UI_BEHAVIOR.md"]
links = 0
for path in paths:
    for target in re.findall(r"\]\(([^)]+)\)", path.read_text(encoding="utf-8")):
        parsed = urlsplit(target.strip("<>"))
        if parsed.scheme or not parsed.path:
            continue
        assert (path.parent / unquote(parsed.path)).exists(), (path, target)
        links += 1
evidence = root / "docs/progress/evidence/2.3"
for width in (768, 1280):
    layout = json.loads((evidence / f"layout-{width}.json").read_text())
    assert layout["width"] == width and layout["height"] == 900

assert len(list(evidence.glob("*.jpg"))) == 6
